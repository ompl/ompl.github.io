54bool ompl::control::PlannerData::removeVertex(const ompl::base::PlannerDataVertex &st)
55{
56 unsigned int index = vertexIndex(st);
57 if (index == INVALID_INDEX)
58 return false;
59
60 std::map<unsigned int, const base::PlannerDataEdge *> edgeMap;
61 getEdges(index, edgeMap);
62
63 for (auto &edgemapit : edgeMap)
64 {
65 // Before deleting the edge, free the control associated with it, if it was decoupled
66 auto *ctrl = const_cast<Control *>(static_cast<const PlannerDataEdgeControl *>(edgemapit.second)->getControl());
67 auto it = decoupledControls_.find(ctrl);
68 if (it != decoupledControls_.end())
69 {
70 siC_->freeControl(*it);
71 decoupledControls_.erase(it);
72 }
73 }
74
75 return ompl::base::PlannerData::removeVertex(index);
76}

83bool ompl::control::PlannerData::removeEdge(const ompl::base::PlannerDataVertex &v1,
84 const ompl::base::PlannerDataVertex &v2)
85{
86 unsigned int index1, index2;
87 index1 = vertexIndex(v1);
88 index2 = vertexIndex(v2);
89
90 if (index1 == INVALID_INDEX || index2 == INVALID_INDEX)
91 return false;
92
93 // Before deleting the edge, free the control associated with it, if it was decoupled
94 auto &edge = static_cast<PlannerDataEdgeControl &>(getEdge(index1, index2));
95 auto *ctrl = const_cast<Control *>(edge.getControl());
96 auto it = decoupledControls_.find(ctrl);
97 if (it != decoupledControls_.end())
98 {
99 siC_->freeControl(*it);
100 decoupledControls_.erase(it);
101 }
102
103 return ompl::base::PlannerData::removeEdge(index1, index2);
104}

114void ompl::control::PlannerData::decoupleFromPlanner()
115{
116 ompl::base::PlannerData::decoupleFromPlanner();
117
118 for (unsigned int i = 0; i < numVertices(); ++i)
119 {
120 for (unsigned int j = 0; j < numVertices(); ++j)
121 {
122 if (edgeExists(i, j))
123 {
124 auto &edge = static_cast<PlannerDataEdgeControl &>(getEdge(i, j));
125 // If this edge's control is not in the decoupled list, clone it and add it
126 auto *ctrl = const_cast<Control *>(edge.getControl());
127 if (decoupledControls_.find(ctrl) == decoupledControls_.end())
128 {
129 Control *clone = siC_->cloneControl(ctrl);
130 decoupledControls_.insert(clone);
131 // Replacing the shallow control pointer with our shiny new clone
132 edge.c_ = clone;
133 }
134 }
135 }
136 }
137}
